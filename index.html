Install

One tag. That's it.

Paste this into your <head>. No npm, no bundler, no setup.

<script src="https://cdn.jsdelivr.net/gh/moistmoi/opalite-ui@main/dist/opalite-ui.js"></script>

<!-- Then write components anywhere -->
<opalite-button variant="primary">Hello</opalite-button>
<opalite-modal title="Welcome">
  <opalite-button slot="trigger">Open</opalite-button>
  Any content here.
</opalite-modal>
